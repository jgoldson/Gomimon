GAME BALANCE CONSTANTS
═══════════════════════════════════════════════════════════════════════════════
HUNGER_DECREASE_RATE          = 1 point per 15 minutes
HUNGER_INCREASE_PER_FEED      = 20 points
GLITCH_INCREASE_PER_FEED      = 5 points
LOW_HUNGER_THRESHOLD          = 30 (show badge)
HIGH_GLITCH_THRESHOLD         = 80 (show warning)
GLITCH_CRASH_THRESHOLD        = 100 (crash state)
FEED_COOLDOWN_MS              = 500ms (min between feeds)
MAX_FEEDS_PER_MINUTE          = 20

BABY_TO_ADULT_FEEDS           = 50 feeds
DIET_DOMINANCE_THRESHOLD      = 0.5 (50%)
